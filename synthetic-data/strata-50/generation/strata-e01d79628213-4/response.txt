    induction s with
    | nil => simp
    | cons x xs ih =>
      intro t hn hs

      have hsub : xs ⊆ u ++ v := by grind
      have hlen := ih (List.nodup_cons.mp hn).2 hsub
      simp only [List.length_cons, List.length_append] at *
      omega
