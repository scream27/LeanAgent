by
  intros α h t l l' inst inst2 H
  induction t generalizing l l' h <;> simp [List.eraseDupsBy.loop] at *
  case nil =>
    apply nodup_swap
    simp
    apply nodup_middle'
    rw [← List.cons_append]
    exact nodup_swap H
  case cons a t ih =>
    by_cases he : a == h
    · rw [eq_of_beq he] at H ⊢
      simp at H
      by_cases hh : h ∈ l
      · simp; exact ih H
      · split <;> first | exact ih H | simp_all
    · split
      · simp_all
        apply ih
        split at H <;> simp_all
        grind
      · simp_all
        have e₁ : a :: h :: (l' ++ l) = (a :: [h]) ++ (l' ++ l) := rfl
        rw [e₁]
        apply ih
        simp
        have e₂ : h :: a :: (l' ++ l) = h :: ((a :: l') ++ l) := rfl
        rw [e₂]
        apply ih
        simp
        grind
